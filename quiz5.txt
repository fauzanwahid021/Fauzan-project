nama = input("Masukkan nama mahasiswa: ")

if nama == "Andi":
    target = 70
elif nama == "Budi":
    target = 75
elif nama == "citra":
    target = 80
else:
    target = 0

if target == 0:
    print("Nama mahasiswa tidak di temukan!")
else:
    print("Target nilai:", target)
    percobaan = 1
    lulus = False
    
while percobaan <= 3:
    nilai = int(input("Masukkan nilai ujian: "))
    
    if nilai >=target:
            lulus = True
            break
    else:
         print("Belum mencapai target!")
         percobaan = percobaan+ 1
            
if lulus == True:
     print("SELAMAT, ANDA LULUS!")
        
else:
     print("ANDA TIDAK LULUS")
